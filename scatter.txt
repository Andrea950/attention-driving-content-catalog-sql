import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

df = pd.read_csv('/content/drive/MyDrive/result_to_plot.csv')

# Log-safe cleaning
df = df[(df["movies"] > 0) & (df["tv_shows"] > 0)]

plt.figure(figsize=(9, 9))
plt.scatter(df["movies"], df["tv_shows"], alpha=0.7)

plt.xscale("log")
plt.yscale("log")

# Range
lo = min(df["movies"].min(), df["tv_shows"].min()) * 0.8
hi = max(df["movies"].max(), df["tv_shows"].max()) * 1.2
plt.xlim(lo, hi)
plt.ylim(lo, hi)
plt.gca().set_aspect("equal", adjustable="box")

# Reference lines
x = np.array([lo, hi])
plt.plot(x, x,     color="black", linestyle="--", linewidth=1, label="Movies = TV Shows")
plt.plot(x, 2 * x, color="gray",  linestyle=":",  linewidth=1, label="TV Shows = 2 × Movies")
plt.plot(x, x / 2, color="gray",  linestyle=":",  linewidth=1, label="Movies = 2 × TV Shows")

# Labels
for _, row in df.iterrows():
    plt.annotate(row["country"], (row["movies"], row["tv_shows"]),
                 xytext=(4, 4), textcoords="offset points", fontsize=8)

plt.title("Movies vs TV Shows by Country (Log Scale)")
plt.xlabel("Number of Movies (log scale)")
plt.ylabel("Number of TV Shows (log scale)")
plt.grid(True, which="both", linestyle="--", linewidth=0.5, alpha=0.3)
plt.legend(frameon=False, loc="upper left")

plt.tight_layout()
plt.savefig("scatter.png", dpi=300, bbox_inches="tight")
plt.show()
